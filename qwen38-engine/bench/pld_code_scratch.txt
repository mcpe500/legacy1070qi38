// http_server.cpp: minimal SSE server for the local engine (no external dependency beyond <sys/socket.h>)
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/socket.h>
#include <netinet/in.h>

struct Client { int fd; char buf[8192]; size_t len; int done; };

static int accept_client(int listen_fd) { return accept(listen_fd, NULL, NULL); }

static void handle_request(struct Client *c) {
    c->done = 1;
    const char *body = "data: {\"text\":\"hello\"}\n\n";
    char hdr[256];
    int n = snprintf(hdr, sizeof(hdr),
        "HTTP/1.1 200 OK\r\nContent-Type: text/event-stream\r\nCache-Control: no-cache\r\n\r\n");
    send(c->fd, hdr, n, 0);
    send(c->fd, body, strlen(body), 0);
}

int serve(int port) {
    int fd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in a;
    memset(&a, 0, sizeof(a));
    a.sin_family = AF_INET;
    a.sin_port = htons(port);
    a.sin_addr.s_addr = INADDR_ANY;
    bind(fd, (struct sockaddr *)&a, sizeof(a));
    listen(fd, 8);
    for (;;) {
        int c = accept_client(fd);
        struct Client cl;
        memset(&cl, 0, sizeof(cl));
        cl.fd = c;
        handle_request(&cl);
        close(c);
    }
    return 0;
}
