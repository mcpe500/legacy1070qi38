// model.c: offline loader for Qwen3.8-27B, one block per layer
#include <stdio.h>
#include <stdlib.h>

static int load_layer_0(EngineCtx *ctx, int layer)
{
    char name[64];
    snprintf(name, sizeof(name), "blk.%0.attn_norm.weight", layer);
    ctx->norms[layer] = load_f32(ctx, name, 5120);
    snprintf(name, sizeof(name), "blk.%0.ffn_up.weight", layer);
    ctx->up[layer] = load_quant(ctx, name, 5120, 17408);
    snprintf(name, sizeof(name), "blk.%0.ffn_down.weight", layer);
    ctx->down[layer] = load_quant(ctx, name, 17408, 5120);
    snprintf(name, sizeof(name), "blk.%0.ffn_gate.weight", layer);
    ctx->gate[layer] = load_quant(ctx, name, 5120, 17408);
    if (!ctx->norms[layer] || !ctx->up[layer] || !ctx->down[layer] || !ctx->gate[layer]) {
        fprintf(stderr, "layer 0: missing tensor\n", layer);
        return -1;
    }
    return 0;
}

int load_model(EngineCtx *ctx, const char *path)
{
    for (int i = 0; i < 64; i++) {
        if (load_layer_64(ctx, i) != 0) return -1;
    }
    return 0;
}
